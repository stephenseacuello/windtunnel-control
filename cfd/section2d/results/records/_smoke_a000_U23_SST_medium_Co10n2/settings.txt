#include "caseParameters"

application     pimpleFoam;

startFrom       latestTime;     // resumes from the last write after an interruption
startTime       0;
stopAt          writeNow;
endTime         $endTime;       // nConv convective times c/U

deltaT          $deltaT0;
adjustTimeStep  yes;
maxCo           $maxCo;
maxDeltaT       $maxDeltaT;

// Fields every nconv/15 c/U (10 c/U by default), keeping the last 3 (restart
// points and the averages);
// the time histories are in postProcessing/.
writeControl    adjustableRunTime;
writeInterval   $writeInterval;
purgeWrite      3;
writeFormat     binary;
writePrecision  8;
writeCompression off;
timeFormat      general;
timePrecision   10;
runTimeModifiable true;

functions
{
    // Section loads. Coefficients: q = 0.5 Uinf^2 (kinematic), A = cRef*dz,
    // Cd along dragDir (= free stream), Cl along liftDir (= z x dragDir).
    // CmPitch is about e2 = liftDir x dragDir = -z through CofR (the section
    // centroid), so CmPitch = -Cm_z; summaries report Cm_z (CCW positive).
    forceCoeffs1
    {
        type            forceCoeffs;
        libs            (forces);
        patches         (blade);
        rho             rhoInf;
        rhoInf          $rhoInf;
        magUInf         $Uinf;
        lRef            $cRef;
        Aref            $Aref;
        CofR            $CofR;
        liftDir         $liftDir;
        dragDir         $dragDir;
        writeControl    runTime;
        writeInterval   $sampleInterval;   // c/U / 50
        log             false;
    }

    // Dimensional loads (N and N m for the cell depth dz; rho = rhoInf),
    // global axes, moment about CofR.
    forces1
    {
        type            forces;
        libs            (forces);
        patches         (blade);
        rho             rhoInf;
        rhoInf          $rhoInf;
        CofR            $CofR;
        writeControl    runTime;
        writeInterval   $sampleInterval;
        log             false;
    }

    // y+ at the first cell centre (min, max, mean over the blade), once per c/U.
    yPlus1
    {
        type            yPlus;
        libs            (fieldFunctionObjects);
        useWallFunction false;
        writeFields     false;
        executeControl  runTime;
        executeInterval $convTime;
        writeControl    runTime;
        writeInterval   $convTime;
        log             false;
    }

    // Wall shear stress every step (needed by the average below); field and
    // min/max written at write times. Separation points come from its mean.
    wallShearStress1
    {
        type            wallShearStress;
        libs            (fieldFunctionObjects);
        patches         (blade);
        writeFields     true;
        executeControl  timeStep;
        executeInterval 1;
        writeControl    writeTime;
        log             false;
    }

    // Time averages over the last nAvg convective times.
    fieldAverage1
    {
        type            fieldAverage;
        libs            (fieldFunctionObjects);
        timeStart       $averageStart;
        restartOnRestart  false;
        restartOnOutput   false;
        periodicRestart   false;
        writeControl    writeTime;
        fields
        (
            U
            {
                mean        on;
                prime2Mean  on;
                base        time;
            }
            p
            {
                mean        on;
                prime2Mean  on;
                base        time;
            }
            wallShearStress
            {
                mean        on;
                prime2Mean  off;
                base        time;
            }
        );
    }

    // Wake probes (shedding frequency) and one upstream probe (free-stream Tu
    // reaching the section). Locations are in caseParameters.
    probes1
    {
        type            probes;
        libs            (sampling);
        fields          (U p k);
        probeLocations  $probeLocations;
        fixedLocations  true;
        writeControl    runTime;
        writeInterval   $sampleInterval;
        log             false;
    }
}

    format      ascii;
    class       dictionary;
    object      fvSchemes;
}
// * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * //

ddtSchemes
{
    // second-order implicit; stays stable where the local Courant number is a few
    default         backward;
}

gradSchemes
{
    default         Gauss linear;
    limitedGrad     cellLimited Gauss linear 1;   // only for the convection reconstruction
}

divSchemes
{
    default         none;
    div(phi,U)      Gauss linearUpwind limitedGrad;
    // k, omega: first-order upwind. With limitedLinear (and linearUpwind) omega
    // went negative near the square corners from the first step at every maxCo
    // tried (bounding in 27-80 % of steps); with upwind it never did (smoke tests, README).
    div(phi,k)      Gauss upwind;
    div(phi,omega)  Gauss upwind;
    // kOmegaSSTLM transport variables: second order, as the T3A tutorial
    div(phi,ReThetat) Gauss linearUpwind limitedGrad;
    div(phi,gammaInt) Gauss linearUpwind limitedGrad;
    div((nuEff*dev2(T(grad(U))))) Gauss linear;
}

laplacianSchemes
{
    // max non-orthogonality of the meshes is about 42 deg (corner cells only)
    default         Gauss linear limited corrected 0.5;
}

interpolationSchemes
{
    default         linear;
}

snGradSchemes
{
    default         limited corrected 0.5;
}

wallDist
{
    method          meshWave;
    correctWalls    true;
}
