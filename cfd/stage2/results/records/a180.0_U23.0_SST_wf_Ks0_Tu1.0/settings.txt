// maxCo 5 (README, smoke test of 7 Oct): the maximum sits in about 100 cells within
// 0.2 mm of the square corners, where the separating shear layer crosses thin
// near-wall cells; in the wake Co <= 0.35 (mean 0.08), so the shedding is resolved

// Fields every nconv/15 c/U (10 c/U by default), keeping the last 3 (restart

functions
{
    // Section loads. Coefficients: q = 0.5 Uinf^2 (kinematic), A = cRef*dz,
    // Cd along dragDir (= free stream), Cl along liftDir (= z x dragDir).
    // CmPitch is about e2 = liftDir x dragDir = -z through CofR (the section
    // centroid), so CmPitch = -Cm_z; summaries report Cm_z (CCW positive).
    forceCoeffs1
    {
        type            forceCoeffs;
        libs            (forces);
        patches         (blade bladeEnds);   // both walls
        rho             rhoInf;
        rhoInf          $rhoInf;
        magUInf         $Uinf;
        lRef            $cRef;
        Aref            $Aref;
        CofR            $CofR;
        liftDir         $liftDir;
        dragDir         $dragDir;
        writeControl    runTime;
        writeInterval   $sampleInterval;   // c/U / 50
        log             false;
    }

    // Dimensional loads (N and N m for the cell depth dz; rho = rhoInf),
    // global axes, moment about CofR.
    forces1
    {
        type            forces;
        libs            (forces);
        patches         (blade bladeEnds);   // both walls
        rho             rhoInf;
        rhoInf          $rhoInf;
        CofR            $CofR;
        writeControl    runTime;
        writeInterval   $sampleInterval;
        log             false;
    }

    // y+ at the first cell centre (min, max, mean over the blade), once per c/U.
    yPlus1
    {
        type            yPlus;
        libs            (fieldFunctionObjects);
        useWallFunction false;
        writeFields     false;
        executeControl  runTime;
        executeInterval $yPlusInterval;   // 1 c/U (shorter for test runs)
        writeControl    runTime;
        writeInterval   $yPlusInterval;
        log             false;
    }

    // Wall shear stress every step (needed by the average below); field and
    // min/max written at write times. Separation points come from its mean.
    wallShearStress1
    {
        type            wallShearStress;
        libs            (fieldFunctionObjects);
        patches         (blade bladeEnds);   // both walls
        writeFields     true;
        executeControl  timeStep;
        executeInterval 1;
        writeControl    writeTime;
        log             false;
    }

    // Time averages over the last nAvg convective times.
    fieldAverage1
    {
        type            fieldAverage;
        libs            (fieldFunctionObjects);
        timeStart       $averageStart;
        restartOnRestart  false;
        restartOnOutput   false;
        periodicRestart   false;
        writeControl    writeTime;
        fields
        (
            U
            {
                mean        on;
                prime2Mean  on;
                base        time;
            }
            p
            {
                mean        on;
                prime2Mean  on;
                base        time;
            }
            wallShearStress
            {
                mean        on;
                prime2Mean  off;
                base        time;
            }
        );
    }

    // Wake probes (shedding frequency) and one upstream probe (free-stream Tu
    // reaching the section). Locations are in caseParameters.
    probes1
    {
        type            probes;
        libs            (sampling);
        fields          (U p k);
        probeLocations  $probeLocations;
        fixedLocations  true;
        writeControl    runTime;
        writeInterval   $sampleInterval;
        log             false;
    }
}

#include "caseParameters"

ddtSchemes
{
    // second-order implicit; stays stable where the local Courant number is a few
    default         backward;
    // gammaInt (kOmegaSSTLM): first-order implicit, which keeps it in [0, 1]. With
    // backward, the impulsive start drove it below 0 in steps 3-5 (min -0.077,
    // 'bounding gammaInt') and above 1 by 1.3e-2, also with a bounded convection
    // scheme; with Euler, no bounding and max 1 + 4e-4 (coarse, alpha 90, 38 m/s;
    // README section 5, tests 6-7). dt is about 1e-3 c/U, so the first-order time error in
    // a source-dominated equation is small.
    ddt(gammaInt)   Euler;
}

gradSchemes
{
    default         Gauss linear;
    limitedGrad     cellLimited Gauss linear 1;   // only for the convection reconstruction
}

divSchemes
{
    default         none;
    div(phi,U)      Gauss linearUpwind limitedGrad;
    // k, omega: first-order upwind. Checked 7 Oct (README): with limitedLinear 1,
    // omega went negative ('bounding omega') in 217 of 224 steps (coarse mesh,
    // alpha 0, 23 m/s, SST); with upwind, in 0 of 1386 steps (medium) and 0 of 377
    // (coarse). The momentum equation stays second order.
    div(phi,k)      Gauss upwind;
    div(phi,omega)  Gauss upwind;
    // kOmegaSSTLM transport variables. ReThetat: second order, as the T3A tutorial.
    // gammaInt (0 <= gamma <= 1): TVD, limited to [0, 1]. With linearUpwind it went
    // below 0 ('bounding gammaInt', min -0.068) in 8 of 380 steps of the 7 Oct LM
    // pipeline test (README section 5, test 3).
    div(phi,ReThetat) Gauss linearUpwind limitedGrad;
    div(phi,gammaInt) Gauss limitedLinear01 1;
    div((nuEff*dev2(T(grad(U))))) Gauss linear;
}

laplacianSchemes
{
    // nonOrthLimit (caseParameters): 0.5 as Stage 1 (max non-orthogonality 39.6 deg medium,

    // every texture case (steep facets; t = 0 too, so t and its reference share the numerics)
    // and for any mesh whose checkMesh maximum exceeds 60 deg.
    default         Gauss linear limited corrected $nonOrthLimit;

RAS
{
    // kOmegaSST (fully turbulent) or kOmegaSSTLM (gamma-ReThetat transition)
    model           $turbulenceModel;
    turbulence      on;
    printCoeffs     on;

    // Only the <model>Coeffs dictionary of the selected model is read (RASModel.C:
    // optionalSubDict(type + "Coeffs")); kOmegaSSTLM derives from kOmegaSST and reads
    // the same decay-control entries from kOmegaSSTLMCoeffs.
    kOmegaSSTCoeffs
    {
        // Free-stream decay control (Spalart and Rumsey 2007; kOmegaSSTBase.C, v2606):
        // sources betaStar*omegaInf*kInf (k) and beta*omegaInf^2 (omega) balance the
        // free-stream destruction, so k and omega keep their far-field values up to
        // the section. With --decay control (default) kInf = kInlet and omegaInf =
        // omegaInlet; with --decay precompensate it is off (kInf = omegaInf = 0).
        decayControl    $freestreamDecayControl;
        kInf            $kAmbient;
        omegaInf        $omegaAmbient;
    }

    kOmegaSSTLMCoeffs
    {
        $kOmegaSSTCoeffs;
        // The lambda (pressure-gradient) iteration for Re_theta,t0 runs to lambdaErr
        // (1e-6) whatever this is; maxLambdaIter (default 10) only sets when the
        // "Number of lambda iterations exceeds maxLambdaIter" warning is printed
        // (kOmegaSSTLM.C, v2606).
        maxLambdaIter   50;
    }
}

== 0/nut (initial and boundary conditions)
==============================================================================
dimensions      [0 2 -1 0 0 0 0];

internalField   uniform $nutInlet;

boundaryField
{
    farfield
    {
        type            calculated;
        value           uniform $nutInlet;
    }
    blade
    {
        // Rough log law (v2606 nutkRoughWallFunctionFvPatchScalarField.C): u* = Cmu^0.25 sqrt(k_P),
        // Ks+ = u* Ks/nu; E is divided by the roughness function fn(Ks+, Cs) for Ks+ > 2.25
        // (transitional form below Ks+ 90, 1 + Cs Ks+ above). No y+ < 11.53 switch: nut_w is
        // limited to 0.5-2x its previous value and so never falls below 0.5 nu.
        // Ks = 0: same condition, smooth (the Stage 2b reference). Valid for Ks below the
        // first-cell-centre height y_p (case.json: yP, KsOverYp).
        type            nutkRoughWallFunction;
        Ks              uniform $KsBlade;       // m, sand-grain height of the painted faces
        Cs              uniform $Cs;            // roughness constant (0.5: uniform sand)
        value           uniform 0;
    }
    bladeEnds
    {
        // square end faces: not painted (no fuzzy skin); Ks = KsEnds (0 by default)
        type            nutkRoughWallFunction;
        Ks              uniform $KsEnds;
        Cs              uniform $Cs;
        value           uniform 0;
    }
    frontAndBack
    {
        type            empty;
    }
}

== 0/omega (initial and boundary conditions)

dimensions      [0 0 -1 0 0 0 0];

internalField   uniform $omegaInlet;

boundaryField
{
    farfield
    {
        type            inletOutlet;
        inletValue      uniform $omegaInlet;
        value           uniform $omegaInlet;
    }
    blade
    {
        // v2606 default blending 'binomial', n 2 (omegaWallFunctionFvPatchScalarField.C: the
        // dictionary constructor's default; the running Stage 1 case writes 'blending binomial;
        // n 2;'): omega_P = sqrt(omega_vis^2 + omega_log^2), omega_vis = 6 nu/(beta1 y^2),
        // omega_log = sqrt(k)/(Cmu^0.25 kappa y). omega_vis dominates on the low-Re texture mesh
        // (y+ < 1, as Stage 1 and T3A); on the wall-function mesh omega_vis/omega_log is about
        // 9.8/y* (0.85 at y* 11.53, 0.25 at y* 40). Unlike 'stepwise', the production G in the
        // wall cell is set from the log law at every y*. Not roughness-aware: roughness enters
        // through nut (nutkRoughWallFunction) and G, which this condition computes from nut_w.
        type            omegaWallFunction;
        value           uniform $omegaInlet;
    }
    bladeEnds
    {
        $blade;
    }
    frontAndBack
    {
        type            empty;
    }
}
